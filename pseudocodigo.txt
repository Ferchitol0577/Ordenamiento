Algoritmo OrdenamientoBurbuja
	
    Definir n, i, j, k, auxiliar Como Entero
    Definir comparaciones, intercambios, pasadas Como Entero
    Definir intercambio Como Logico
    Definir numeros Como Entero
	
    // Creamos el arreglo para guardar los números
    n <- 5
    Dimension numeros[n]
	
    numeros[1] <- 34
    numeros[2] <- 12
    numeros[3] <- 25
    numeros[4] <- 8
    numeros[5] <- 19
	
    // Inicializamos las variables
    i <- 0
    intercambio <- Verdadero
    comparaciones <- 0
    intercambios <- 0
    pasadas <- 0
	
    // Repetimos mientras haya números por ordenar
    Mientras i < n - 1 Y intercambio = Verdadero Hacer
		
        // Indicar que todavía no hubo intercambios
        intercambio <- Falso
		
        // Compararamos los números del arreglo
        Para j <- 1 Hasta n - 1 - i Hacer
			
            comparaciones <- comparaciones + 1
			
            // Verificamos si el número actual es mayor al siguiente
            Si numeros[j] > numeros[j + 1] Entonces
				
                // Intercambia los números de posición
                auxiliar <- numeros[j]
                numeros[j] <- numeros[j + 1]
                numeros[j + 1] <- auxiliar
				
                intercambios <- intercambios + 1
				
                // Indicamos que se realizó un intercambio
                intercambio <- Verdadero
				
            FinSi
			
        FinPara
		
        // Pasa a la siguiente vuelta
        i <- i + 1
        pasadas <- pasadas + 1
		
        Escribir "Pasada ", pasadas, ":"
		
        Para k <- 1 Hasta n Hacer
            Escribir Sin Saltar numeros[k], " "
        FinPara
		
        Escribir ""
		
    FinMientras
	
    Escribir ""
    Escribir "Arreglo ordenado:"
	
    Para k <- 1 Hasta n Hacer
        Escribir Sin Saltar numeros[k], " "
    FinPara
	
    Escribir ""
    Escribir "Total de comparaciones: ", comparaciones
    Escribir "Total de intercambios: ", intercambios
    Escribir "Total de pasadas: ", pasadas
	
FinAlgoritmo
